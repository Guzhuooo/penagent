"""Generate tiny 48x48 flat PNG icons for the penagent mini-app and inline them
into app/src/assets/icons.js as base64 data URIs.

Why inline: on the Falcon/QuickJS toolchain only package.json icon.png is copied
into the package; loose files under src/assets/ are not packaged, so images must
travel as data URIs. Device fonts cannot render emoji glyphs.

Flat geometry only: drawn on a 4x supersample grid (192x192), LANCZOS down to
48x48, then snapped to the design palette + 1-bit alpha so each PNG stays far
below the 2 KiB data URI budget.
Run with the bundled Python + Pillow.
"""
import base64
import io
import os

from PIL import Image, ImageDraw

S = 192  # 48 * 4 supersample grid; all shape coordinates below are in this space
BRAND = (0x58, 0x8B, 0xD8, 255)       # styles/var.less @brand
SECONDARY = (0x8D, 0x99, 0xA6, 255)   # styles/var.less @text-secondary
PRIMARY = (0xE9, 0xEE, 0xF4, 255)     # styles/var.less @text-primary


def finalize(image, palette, cut=96):
    """Supersample -> 48x48 -> nearest palette colour + 1-bit alpha."""
    image = image.resize((48, 48), Image.LANCZOS)
    pixels = image.load()
    for y in range(48):
        for x in range(48):
            r, g, b, a = pixels[x, y]
            if a < cut:
                pixels[x, y] = (0, 0, 0, 0)
                continue
            best = min(palette, key=lambda c: (c[0] - r) ** 2 + (c[1] - g) ** 2 + (c[2] - b) ** 2)
            pixels[x, y] = (best[0], best[1], best[2], 255)
    return image


def check_on():
    image = Image.new('RGBA', (S, S), (0, 0, 0, 0))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((16, 16, 176, 176), radius=36, fill=BRAND)
    draw.line([(52, 96), (80, 124), (136, 64)], fill=PRIMARY, width=20, joint='curve')
    return finalize(image, [BRAND, PRIMARY])


def check_off():
    image = Image.new('RGBA', (S, S), (0, 0, 0, 0))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((16, 16, 176, 176), radius=36, outline=SECONDARY, width=14)
    return finalize(image, [SECONDARY])


def plus():
    image = Image.new('RGBA', (S, S), (0, 0, 0, 0))
    draw = ImageDraw.Draw(image)
    draw.rounded_rectangle((44, 88, 148, 104), radius=8, fill=BRAND)
    draw.rounded_rectangle((88, 44, 104, 148), radius=8, fill=BRAND)
    return finalize(image, [BRAND])


ICONS = [('check-on', check_on), ('check-off', check_off), ('plus', plus)]
OUT = os.path.join('D:/penagent/app/src/assets/icons.js')
os.makedirs(os.path.dirname(OUT), exist_ok=True)

lines = [
    '// ASCII-only PNG icons inlined as data URIs.',
    '//',
    '// Why inline: the Falcon/QuickJS packer only copies package.json icon.png,',
    '// so loose images under src/assets/ never reach the package. Inline PNGs are',
    '// referenced from templates via <image :src="icon(name)" mode="aspectFit">',
    '// with explicit vw/vh class geometry. Device fonts cannot render emoji.',
    '// Generated by Pillow at 48x48 (4x supersampled, LANCZOS, palette + 1-bit',
    '// alpha); do not hand edit.',
    'export const ICONS = {',
]
report = []
for name, fn in ICONS:
    image = fn()
    buffer = io.BytesIO()
    image.save(buffer, format='PNG', optimize=True)
    payload = buffer.getvalue()
    uri = 'data:image/png;base64,' + base64.b64encode(payload).decode('ascii')
    assert payload[:8] == b'\x89PNG\r\n\x1a\n', name
    assert len(uri) < 2048, (name, len(uri))
    lines.append("  '" + name + "': '" + uri + "',")
    report.append((name, len(payload), len(uri)))
lines += [
    '};',
    '',
    '// Return the data URI, or empty string when the name is unknown so callers',
    '// can fall back to text instead of crashing.',
    'export function icon(name) {',
    "  return ICONS[String(name || '')] || '';",
    '}',
]
text = '\n'.join(lines) + '\n'
assert all(ord(ch) < 128 for ch in text), 'icons.js must stay ASCII-only'
with open(OUT, 'w', encoding='ascii', newline='\n') as handle:
    handle.write(text)
print('wrote', OUT, 'bytes', len(text))
for name, raw, uri in report:
    print(name, 'png_bytes=' + str(raw), 'uri_chars=' + str(uri))

# Visual preview: 6x nearest + device-size 22px sample, on @bg-muted.
preview = Image.new('RGBA', (360, 210), (0x1F, 0x26, 0x30, 255))
x = 16
for name, fn in ICONS:
    icon = fn()
    preview.alpha_composite(icon.resize((96, 96), Image.NEAREST), (x, 16))
    preview.alpha_composite(icon.resize((22, 22), Image.LANCZOS), (x + 32, 140))
    x += 112
preview.save(os.path.join(os.environ['TEMP'], 'penagent-icons', 'preview.png'))
print('preview saved')
